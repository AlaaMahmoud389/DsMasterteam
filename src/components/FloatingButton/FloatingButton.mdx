    ['Pill — Small',      <FloatingButton key="ps" variant="brand" size="sm" iconOnly={false} leadIcon={<PlusIcon />}>Button</FloatingButton>],
    ['Neutral icon',      <FloatingButton key="n"  variant="neutral" size="lg" aria-label="Add"><PlusIcon /></FloatingButton>],
    ['Secondary icon',    <FloatingButton key="s"  variant="secondary" size="lg" aria-label="Add"><PlusIcon /></FloatingButton>],
  ].map(([label, btn]) => (
    <div key={label} style={{ background: '#fff', border: '1px solid #e5e7eb', borderRadius: 10, padding: '18px 16px 14px', display: 'flex', flexDirection: 'column', gap: 11 }}>
      <span style={{ fontSize: 11, color: '#6b7280', fontWeight: 600 }}>{label}</span>
      {btn}
    </div>
  ))}
</div>

| Mode | `iconOnly` | Shape | When to use |
|---|---|---|---|
| Circle | `true` (default) | Square aspect ratio, round corners | Primary FAB, single action |
| Pill | `false` | Auto width, round corners | When label adds clarity |

```jsx
// Icon-only (circle) — always provide aria-label
<FloatingButton variant="brand" aria-label="Create new item">
  <PlusIcon />
</FloatingButton>

// Pill mode — label replaces aria-label requirement
<FloatingButton variant="brand" iconOnly={false} leadIcon={<PlusIcon />}>
  Create
</FloatingButton>
```

## Implementation Examples

```jsx
import { FloatingButton } from '@/components/FloatingButton';
import { PlusIcon, ArrowRightIcon } from '@/components/Button/ButtonIcons';

// Brand FAB — primary action (most common)
<FloatingButton variant="brand" aria-label="Create new item">
  <PlusIcon />
</FloatingButton>

// Neutral FAB — secondary action
<FloatingButton variant="neutral" size="sm" aria-label="Add">
  <PlusIcon />
</FloatingButton>

// Secondary FAB — softer brand tint
<FloatingButton variant="secondary" aria-label="Add">
  <PlusIcon />
</FloatingButton>

// Pill mode — icon + label
<FloatingButton variant="brand" iconOnly={false} leadIcon={<PlusIcon />}>
  Create
</FloatingButton>

// On colored background
<FloatingButton variant="brand" onColor aria-label="Add">
  <PlusIcon />
</FloatingButton>

// Disabled
<FloatingButton variant="brand" disabled aria-label="Add (unavailable)">
  <PlusIcon />
</FloatingButton>

// RTL pill
<FloatingButton variant="brand" dir="rtl" iconOnly={false}
  leadIcon={<ArrowRightIcon style={{ transform: 'scaleX(-1)' }} />}>
  إضافة
</FloatingButton>
```
