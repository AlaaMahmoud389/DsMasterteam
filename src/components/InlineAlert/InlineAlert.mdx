## Implementation Examples

```jsx
import { InlineAlert } from '@/components/InlineAlert';

// Info — neutral, dismissible
<InlineAlert
  type="info"
  title="Scheduled maintenance tonight."
  description="The service will be unavailable from 00:00–02:00 UTC. No action required."
  onClose={() => setShowInfo(false)}
/>

// Success — confirmation, non-dismissible
<InlineAlert
  type="success"
  title="Profile updated successfully."
  dismissible={false}
/>

// Warning — with action buttons
<InlineAlert
  type="warning"
  title="Your subscription expires in 3 days."
  description="Renew now to avoid service interruption."
  showActions
  primaryActionLabel="Renew Now"
  secondaryActionLabel="Remind Me"
  onPrimaryAction={() => navigate('/billing')}
  onSecondaryAction={() => scheduleReminder()}
  onClose={() => setShowWarning(false)}
/>

// Destructive — blocking error, no dismiss
<InlineAlert
  type="destructive"
  title="Payment failed."
  description="Check your card details and try again."
  dismissible={false}
  showActions
  primaryActionLabel="Update Payment"
  secondaryActionLabel="Cancel"
  onPrimaryAction={() => navigate('/payment')}
/>

// Mobile layout
<InlineAlert
  type="info"
  mobile
  title="New update available."
  description="Refresh the page to get the latest version."
  onClose={() => setShowAlert(false)}
/>

// RTL Arabic
<InlineAlert
  type="warning"
  rtl
  title="ستنتهي صلاحية اشتراكك خلال 3 أيام."
  description="جدد الاشتراك الآن لتجنب انقطاع الخدمة."
  showActions
  primaryActionLabel="تجديد الآن"
  secondaryActionLabel="تذكيري لاحقاً"
  onClose={() => setShowWarning(false)}
/>

// RTL Mobile
<InlineAlert
  type="success"
  mobile
  rtl
  title="تم تحديث الملف الشخصي بنجاح."
  dismissible={false}
/>

// Controlled visibility pattern
const [visible, setVisible] = useState(true);
{visible && (
  <InlineAlert
    type="destructive"
    title="Form has errors."
    description="Please fix the highlighted fields before submitting."
    dismissible={false}
  />
)}
```
