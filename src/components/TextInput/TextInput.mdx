## Usage

```jsx
import { TextInput } from './TextInput';

// Controlled interactive input
const [value, setValue] = useState('');
<TextInput
  label="Email"
  placeholder="Enter your email"
  value={value}
  onChange={(e) => setValue(e.target.value)}
  showHelperText
  helperText="We'll never share your email"
/>

// Required field with semibold label
<TextInput label="Full Name" labelType="semibold" required placeholder="John Doe" />

// Error state
<TextInput
  label="Password"
  error
  placeholder="Enter password"
  showHelperText
  helperText="Password must be at least 8 characters"
/>

// Large with leading icon
<TextInput size="large" showIcon label="Search" placeholder="Search..." />

// Read-only
<TextInput state="read-only" label="Username" defaultValue="faisal.rashid" />

// Disabled
<TextInput state="disabled" label="Email" defaultValue="user@example.com" />

// RTL
<TextInput rtl label="البريد الإلكتروني" placeholder="أدخل بريدك الإلكتروني" />
```
