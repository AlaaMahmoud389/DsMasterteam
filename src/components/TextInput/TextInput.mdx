{/* ── Usage Guidelines ───────────────────────────────────────────── */}
<h2 id="usage" style={{ fontSize: 20, fontWeight: 700, color: '#000b36', marginBottom: 8 }}>Usage Guidelines</h2>

<div style={{ display: 'grid', gridTemplateColumns: '1fr 1fr', gap: 20, marginBottom: 28 }}>
  <div style={{ border: '1.5px solid #d1fae5', borderRadius: 8, padding: '16px 18px', background: '#f0fdf4' }}>
    <div style={{ fontSize: 13, fontWeight: 700, color: '#065f46', marginBottom: 10 }}>Do</div>
    <ul style={{ margin: 0, paddingLeft: 18, fontSize: 13, color: '#374151', lineHeight: 1.8 }}>
      <li>Always provide a visible label or <code>aria-label</code></li>
      <li>Use <code>helperText</code> to explain the expected format</li>
      <li>Turn on <code>error={'{true}'}</code> and set <code>helperText</code> to the error message</li>
      <li>Use size <strong>large</strong> for touch-first or prominent form fields</li>
      <li>Pair prefix/suffix with a clear text label or recognisable icon</li>
    </ul>
  </div>
  <div style={{ border: '1.5px solid #fee2e2', borderRadius: 8, padding: '16px 18px', background: '#fff5f5' }}>
    <div style={{ fontSize: 13, fontWeight: 700, color: '#991b1b', marginBottom: 10 }}>Don't</div>
    <ul style={{ margin: 0, paddingLeft: 18, fontSize: 13, color: '#374151', lineHeight: 1.8 }}>
      <li>Don't use TextInput for multi-line text — use Textarea</li>
      <li>Don't rely solely on placeholder text as a label</li>
      <li>Don't show an error without a descriptive <code>helperText</code> message</li>
      <li>Don't mix prefix and suffix types that confuse the action (e.g. two dropdowns)</li>
      <li>Don't disable a field without providing an explanation nearby</li>
    </ul>
  </div>
</div>
