## Code Examples

```jsx
import { Chip, ChipGroup } from '@/components/Chip';

// Basic chip
<Chip label="Design" chipStyle="primary" size="lg" rounded />

// Toggled / selectable chip
<Chip
  label="UX Research"
  chipStyle="primary"
  size="lg"
  rounded
  selected={isSelected}
  onClick={() => setIsSelected(!isSelected)}
/>

// Dismissible chip
<Chip
  label="Marketing"
  chipStyle="neutral"
  size="lg"
  rounded
  showCloseButton
  onClose={() => removeTag('Marketing')}
/>

// With lead + trail icons
<Chip label="Category" chipStyle="primary" size="lg" rounded showLeadIcon showTrailIcon />

// RTL with Arabic label
<Chip label="Item" labelAr="اختيار" dir="rtl" chipStyle="primary" size="lg" rounded />

// On-color context
<div style={{ background: '#194AAA', padding: 16 }}>
  <Chip label="Featured" chipStyle="primary" size="lg" rounded onColor />
</div>

// ChipGroup
<ChipGroup
  items={[
    { label: 'Design' },
    { label: 'Development', selected: true },
    { label: 'Research', onClose: () => remove('Research') },
  ]}
  chipStyle="primary"
  size="lg"
  rounded
  showCloseButton
/>
```
