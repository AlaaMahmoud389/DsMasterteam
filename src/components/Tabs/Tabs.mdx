## Usage

```jsx
import { VerticalTab, HorizontalTab, HorizontalMoreTab, VerticalTabList, HorizontalTabList } from './Tabs';

// Interactive vertical navigation
const [active, setActive] = useState('tab1');
<VerticalTabList
  tabs={[
    { id: 'tab1', label: 'Overview' },
    { id: 'tab2', label: 'Details' },
    { id: 'tab3', label: 'Settings', disabled: true },
  ]}
  activeTab={active}
  onTabChange={setActive}
  size="medium"
/>

// Interactive horizontal navigation with divider
<HorizontalTabList
  tabs={tabs}
  activeTab={active}
  onTabChange={setActive}
  size="large"
  divider
/>

// Flush horizontal tabs (no leading padding)
<HorizontalTabList tabs={tabs} activeTab={active} onTabChange={setActive} flush />

// Single tab — presentational, all states
<VerticalTab size="large" label="Selected" selected />
<HorizontalTab size="medium" label="Hovered" state="hovered" />

// RTL
<VerticalTabList rtl tabs={arabicTabs} activeTab={active} onTabChange={setActive} />
<HorizontalTabList rtl tabs={arabicTabs} activeTab={active} onTabChange={setActive} />
```
