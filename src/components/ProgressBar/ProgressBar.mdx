## Implementation Examples

```jsx
import { ProgressBar } from '@/components/ProgressBar';

// Basic controlled progress
const [progress, setProgress] = useState(0);
<ProgressBar
  value={progress}
  label="Uploading file"
  helperText={`${progress}% complete`}
/>

// With status lifecycle
const [progress, setProgress] = useState(0);
const [status, setStatus] = useState('default');

// On upload tick:
setProgress(prev => {
  const next = Math.min(prev + 10, 100);
  if (next === 100) setStatus('success');
  return next;
});

<ProgressBar
  value={progress}
  status={status}
  label="Uploading report.pdf"
  helperText={`${progress}% - ${Math.round((100 - progress) / 10)} seconds remaining`}
/>

// Error state
<ProgressBar
  value={partialProgress}
  status="error"
  label="Upload failed"
  helperText="Network error - please retry"
/>

// Neutral variant for background sync
<ProgressBar
  variant="Neutral"
  value={syncProgress}
  label="Syncing data"
  helperText="Fetching records..."
/>

// Sizes
<ProgressBar size="Small"  value={60} label="Small" />
<ProgressBar size="Medium" value={60} label="Medium" />
<ProgressBar size="Large"  value={60} label="Large" />

// RTL Arabic
<ProgressBar
  rtl={true}
  value={75}
  label="تحميل الملف"
  helperText="٧٥٪ مكتمل"
/>

// Bar only (no label/helper - for compact table cells)
<ProgressBar
  showLabel={false}
  showHelperText={false}
  value={row.completionPct}
  aria-label={`${row.name} - ${row.completionPct}% complete`}
/>
```
