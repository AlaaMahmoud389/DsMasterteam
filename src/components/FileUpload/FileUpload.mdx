<FileUploadMultiple
  hint="Maximum file size allowed is 2MB, supported file formats include .jpg, .png, and .pdf."
  multiple
  files={files}
  onChange={(fileList) => {
    const entries = Array.from(fileList).map((f) => ({
      id: f.name + Date.now(),
      name: f.name,
      status: 'uploaded',
    }));
    setFiles((prev) => [...prev, ...entries]);
  }}
  onRemove={(id) => setFiles((prev) => prev.filter((f) => f.id !== id))}
/>`}</pre>
  </div>
