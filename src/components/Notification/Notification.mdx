<div style={{ display: 'grid', gridTemplateColumns: '1fr 1fr', gap: 14, marginBottom: 16 }}>
  <div style={{ background: '#fff', border: '1px solid #e5e7eb', borderRadius: 10, padding: 16 }}>
    <div style={{ fontSize: 11, fontWeight: 700, textTransform: 'uppercase', letterSpacing: '.08em', color: '#6b7280', marginBottom: 12 }}>Dismissible (default)</div>
    <Notification variant="Warning" dismissible={true} />
  </div>
  <div style={{ background: '#fff', border: '1px solid #e5e7eb', borderRadius: 10, padding: 16 }}>
    <div style={{ fontSize: 11, fontWeight: 700, textTransform: 'uppercase', letterSpacing: '.08em', color: '#6b7280', marginBottom: 12 }}>Non-Dismissible</div>
    <Notification variant="Warning" dismissible={false} />
  </div>
</div>

| dismissible | Behaviour |
|---|---|
| `true` (default) | Shows × close button · calls `onDismiss` when clicked |
| `false` | No close button · notification is persistent |

Use `dismissible={false}` for critical errors that require user action before the page can continue.

## Implementation Examples

```jsx
import { Notification } from '@/components/Notification';

// Critical — blocking error (non-dismissible)
<Notification
  variant="Critical"
  dismissible={false}
  title="Error:"
  message="Payment failed. Please check your card details and try again."
/>

// Warning — cautionary, dismissible
<Notification
  variant="Warning"
  title="Warning:"
  message="Your subscription expires in 3 days. Renew now to avoid service interruption."
  onDismiss={() => setShowWarning(false)}
/>

// Success — confirmation
<Notification
  variant="Success"
  title="Success:"
  message="Your profile has been updated successfully."
  onDismiss={() => setShowSuccess(false)}
/>

// Info — informational
<Notification
  variant="Info"
  title="Info:"
  message="Scheduled maintenance on Saturday 00:00–02:00 UTC. No action required."
  onDismiss={() => setShowInfo(false)}
/>

// Neutral — system message
<Notification
  variant="Neutral"
  title="Note:"
  message="Some features may be unavailable in your region."
  onDismiss={() => setShowNote(false)}
/>

// RTL Arabic
<Notification
  variant="Critical"
  rtl={true}
  title="خطأ:"
  message="فشل الدفع. يرجى التحقق من بيانات البطاقة والمحاولة مرة أخرى."
  onDismiss={() => setShowError(false)}
/>

// Controlled visibility pattern
const [visible, setVisible] = useState(true);
{visible && (
  <Notification
    variant="Warning"
    title="Warning:"
    message="Unsaved changes will be lost."
    onDismiss={() => setVisible(false)}
  />
)}
```
