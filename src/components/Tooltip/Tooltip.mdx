## Implementation Examples

```jsx
import { Tooltip } from '@/components/Tooltip';

// Basic tooltip — light theme, no beak
<Tooltip
  title="Tooltip title"
  body="Max width of tooltips is 240px - text will wrap automatically"
/>

// Dark theme tooltip with top beak centered
<Tooltip
  inverted={true}
  beakPlacement="Top"
  beakAlignment="Center"
  title="Dark tooltip"
  body="Use dark theme over light page content for better contrast."
/>

// Without the feedback icon
<Tooltip
  icon={false}
  title="No icon"
  body="The icon is optional — disable it for a more compact appearance."
/>

// RTL Arabic
<Tooltip
  rtl={true}
  inverted={true}
  beakPlacement="Bottom"
  beakAlignment="Center"
  title="عنوان التلميح"
  body="أقصى عرض للتلميحات هو 240px - سيتم محاذاة النص بشكل تلقائي"
/>

// Anchored to a trigger with correct ARIA wiring
<div style={{ position: 'relative', display: 'inline-block' }}>
  <button
    aria-describedby="my-tooltip"
    onMouseEnter={() => setVisible(true)}
    onMouseLeave={() => setVisible(false)}
    onFocus={() => setVisible(true)}
    onBlur={() => setVisible(false)}
  >
    Hover or focus me
  </button>
  {visible && (
    <div
      id="my-tooltip"
      role="tooltip"
      style={{ position: 'absolute', bottom: '100%', left: '50%', transform: 'translateX(-50%)', marginBottom: 8 }}
    >
      <Tooltip
        beakPlacement="Bottom"
        beakAlignment="Center"
        inverted={true}
        title="Action info"
        body="This action will save your changes."
      />
    </div>
  )}
</div>
```
