## Implementation Examples

```jsx
import { Button } from '@/components/Button';

// Primary CTA
<Button variant="primary" leadIcon={<ArrowRightIcon />} onClick={handleSubmit}>
  Submit
</Button>

// Secondary action
<Button variant="secondary-solid" size="md">Cancel</Button>

// Danger — irreversible
<Button variant="danger" leadIcon={<TrashIcon />} onClick={handleDelete}>
  Delete account
</Button>

// Danger secondary — softer
<Button variant="danger-secondary" leadIcon={<TrashIcon />}>Remove</Button>

// Loading
<Button variant="primary" loading>Saving...</Button>

// Icon only
<Button iconOnly aria-label="Add item" size="lg"><PlusIcon /></Button>

// RTL Arabic
<Button variant="primary" dir="rtl"
  leadIcon={<ArrowRightIcon style={{ transform: 'scaleX(-1)' }} />}>
  الخطوة التالية
</Button>
```
