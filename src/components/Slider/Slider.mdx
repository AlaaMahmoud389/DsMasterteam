import { Meta } from '@storybook/addon-docs/blocks';
import React, { useState } from 'react';
import { Slider } from './Slider';

export const SingleDemo = () => {
  const [val, setVal] = useState(20);
  return (
    <div style={{ padding: '16px 0', fontFamily: 'IBM Plex Sans Arabic, sans-serif' }}>
      <Slider value={val} onChange={setVal} label="Label" helperText="Help Text" />
      <div style={{ marginTop: 8, fontSize: 12, color: '#6b7280' }}>Value: {val}%</div>
    </div>
  );
};

export const RangeDemo = () => {
  const [range, setRange] = useState({ min: 20, max: 70 });
  return (
    <div style={{ padding: '16px 0', fontFamily: 'IBM Plex Sans Arabic, sans-serif' }}>
      <Slider
        range
        minValue={range.min} maxValue={range.max} onRangeChange={setRange}
        label="Label" helperText="Help Text"
      />
      <div style={{ marginTop: 8, fontSize: 12, color: '#6b7280' }}>Range: {range.min}% – {range.max}%</div>
    </div>
  );
};

## Usage

```jsx
import { Slider } from './Slider';

// Single interactive slider
const [value, setValue] = useState(20);
<Slider value={value} onChange={setValue} label="Volume" />

// Range slider
const [range, setRange] = useState({ min: 20, max: 70 });
<Slider
  range
  minValue={range.min}
  maxValue={range.max}
  onRangeChange={setRange}
  label="Price range"
/>

// RTL
<Slider value={20} rtl label="عنوان" helperText="نص مساعد" />

// Small size, semibold label, required
<Slider size="small" value={50} label="Amount" labelType="semibold" required />

// Read-only display (no onChange)
<Slider value={75} showHelperText={false} />
```
