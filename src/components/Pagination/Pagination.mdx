## Implementation Examples

```jsx
import { Pagination } from '@/components/Pagination';

// Basic controlled usage
const [page, setPage] = useState(1);
<Pagination
  size="Medium"
  currentPage={page}
  totalPages={50}
  onPageChange={(p) => setPage(p)}
/>

// With data fetch on page change
const [page, setPage] = useState(1);
const { data, isLoading } = useQuery(['items', page], () => fetchItems(page));

<Pagination
  size="Medium"
  currentPage={page}
  totalPages={data?.totalPages ?? 1}
  onPageChange={(p) => setPage(p)}
/>

// Large size for mobile
<Pagination
  size="Large"
  currentPage={page}
  totalPages={20}
  onPageChange={setPage}
/>

// RTL Arabic
<Pagination
  size="Medium"
  rtl={true}
  currentPage={page}
  totalPages={20}
  onPageChange={setPage}
/>

// URL-synced pagination (React Router / Next.js)
const [searchParams, setSearchParams] = useSearchParams();
const page = Number(searchParams.get('page') ?? 1);

<Pagination
  size="Medium"
  currentPage={page}
  totalPages={totalPages}
  onPageChange={(p) => setSearchParams({ page: p })}
/>

// Conditional render — hide when only 1 page
{totalPages > 1 && (
  <Pagination
    currentPage={page}
    totalPages={totalPages}
    onPageChange={setPage}
  />
)}
```
